  1 library ieee;
  2   use ieee.std_logic_1164.all;
  3
  4 entity fifo_sync is
  5   generic (
  6     G_WIDTH : positive := 8;
  7     G_DEPTH : positive := 16
  8   );
  9   port (
 10     clk       : in  std_logic;
 11     rst_n     : in  std_logic;
 12     wr_en_i   : in  std_logic;
 13     wr_data_i : in  std_logic_vector(G_WIDTH - 1 downto 0);
 14     full_o    : out std_logic;
 15     rd_en_i   : in  std_logic;
 16     rd_data_o : out std_logic_vector(G_WIDTH - 1 downto 0);
 17     empty_o   : out std_logic
 18   );
 19 end entity fifo_sync;
 20
 21 architecture rtl of fifo_sync is
 22
 23   type t_memory is array (0 to G_DEPTH - 1) of std_logic_vector(G_WIDTH - 1 downto 0);
 24
 25   signal memory_r : t_memory;
 26
 27   signal wr_ptr_r : natural range 0 to G_DEPTH - 1 := 0;
 28   signal rd_ptr_r : natural range 0 to G_DEPTH - 1 := 0;
 29   signal count_r  : natural range 0 to G_DEPTH := 0;
 30
 31   signal wr_go : boolean;
 32   signal rd_go : boolean;
 33
 34 begin
 35
 36   -- A write to a full FIFO, or a read from an empty one, is
 37   -- ignored: a misbehaving master must not desynchronise it.
 38   wr_go <= wr_en_i = '1' and count_r < G_DEPTH;
 39   rd_go <= rd_en_i = '1' and count_r > 0;
 40
 41   -- No reset here: a reset on every element prevents block RAM
 42   -- inference on most tools.
 43   p_memory : process (clk) is
 44   begin
 45     if rising_edge(clk) then
 46       if wr_go then
 47         memory_r(wr_ptr_r) <= wr_data_i;
 48       end if;
 49     end if;
